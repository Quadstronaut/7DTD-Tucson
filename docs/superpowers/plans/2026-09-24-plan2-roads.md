# Plan 2 — Real Roads + Direct World Assembly Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Real OSM roads (I-10, I-19, arterials, collectors) graded into the terrain and painted as asphalt, written straight into a playable world folder `Tucson_AZ` without re-running the mod.

**Architecture:** `roads.py` fetches/caches OSM highway ways, warps them, and produces (a) a road mask for splat3 and (b) graded terrain: each way's terrain profile is sampled along its centerline, 1-D smoothed, and spread across the road width via nearest-centerline lookup (`distance_transform_edt(return_indices=True)`), blended over a feather. `assemble.py` copies the never-loaded engine files from the RWG shell (Tisuviro County, Towns/Wilderness None) and writes our `dtm.raw`, `splat3.png`, `prefabs.xml`, `spawnpoints.xml` into `GeneratedWorlds/Tucson_AZ`.

**Tech Stack:** as Plan 1 (numpy, Pillow, scipy, pytest, stdlib).

**Spec:** `docs/specs/2026-09-24-tucson-map-design.md` (§13, Plan 1 RESULT).

## Global Constraints

- `dtm.raw`: LE uint16, 10240², **row 0 = south**, value = round(blocks × 256).
- `splat3.png`: RGBA 10240², **row 0 = north**, asphalt = (255,0,0,255), gravel = (0,255,0,255), empty = (0,0,0,0). Hard edges (vanilla uses only pure values).
- Prefab Y = dtm + YOffset + 1; prefab x/z = pixel − 5120 (z from dtm row, i.e. south-origin).
- Shell engine files copied verbatim: `main.ttw`, `map_info.xml`, `biomes.png`, `splat4.png`, `radiation.png`. **Never copy** `*_processed*`, `*_half*`, `checksums.txt`, `*.orig`, `map_preview.png`.
- Whether the game rebuilds derived files for a hand-assembled world is UNVERIFIED → Task 4 checks it; fallback = feed graded heightmap through the mod.
- v1 scope: bridges/overpasses graded at-grade (no bridge prefabs yet); no residential streets (tiles come in Plan 4).

## Review Focus

1. Dual-carriageway motorways are two OSM ways — both must be painted and graded, not averaged into a ridge between them. (Task 2 test: two parallel ways stay flat across.)
2. Roads leaving the box must be clipped, not smeared along the edge. (Task 1 test.)
3. Intersections where ways of different profiles meet must not create cliffs > 2 blocks between adjacent road pixels. (Task 2 test.)
4. Road grading must not lower a flattened landmark zone's surface noticeably (zones are already flat; profile smoothing keeps them). (Task 2 test.)
5. Assembly must refuse to run if the shell folder has been loaded (derived files present) unless `--allow-loaded`. (Task 3 test.)

```python
"""Real OSM roads -> graded terrain + splat3 asphalt mask."""
import json, os
import numpy as np
from scipy import ndimage
from PIL import Image, ImageDraw
from . import zones
from .config import ROOT

CACHE = os.path.join(ROOT, "cache", "osm", "roads.json")


def classes(cfg):
    return [k for k, v in cfg["roads"].items() if isinstance(v, list)]


def fetch_ways(cfg):
    if os.path.exists(CACHE):
        return json.load(open(CACHE))
    b = cfg["box"]; cls = "|".join(classes(cfg))
    d = zones.overpass(f'way["highway"~"^({cls})$"]({b["south"]},{b["west"]},{b["north"]},{b["east"]});out geom tags;')
    ways = [{"id": e["id"], "cls": e["tags"]["highway"], "bridge": e["tags"].get("bridge", "no") != "no",
             "pts": [(p["lat"], p["lon"]) for p in e["geometry"]]}
            for e in d["elements"] if e["type"] == "way" and "geometry" in e]
    if not ways:
        raise RuntimeError("no roads returned from OSM")
    os.makedirs(os.path.dirname(CACHE), exist_ok=True); json.dump(ways, open(CACHE, "w"))
    return ways


def to_world(ways, warp):
    """Warp to (x, row-from-north); drop vertices outside the box (box edges clamp to the border,
    so a run of clamped vertices would smear along the edge)."""
    b = warp.box; out = []
    for w in ways:
        p = np.array(w["pts"], float)
        inside = (p[:, 0] >= b["south"]) & (p[:, 0] <= b["north"]) & (p[:, 1] >= b["west"]) & (p[:, 1] <= b["east"])
        # keep maximal inside runs as separate pieces
        idx = np.flatnonzero(inside)
        if len(idx) < 2:
            continue
        for run in np.split(idx, np.flatnonzero(np.diff(idx) > 1) + 1):
            if len(run) < 2:
                continue
            q = p[run]
            xy = np.stack([warp.x(q[:, 1]), warp.z(q[:, 0])], 1)
            xy = np.clip(xy, 0, warp.N - 1)
            out.append({**w, "xy": xy})
    return out
```

```python
def densify(xy, step=1.0):
    seg = np.diff(xy, axis=0); L = np.hypot(*seg.T)
    pts = [xy[:1]]
    for a, s, l in zip(xy[:-1], seg, L):
        n = max(1, int(np.ceil(l / step)))
        pts.append(a + s * (np.arange(1, n + 1)[:, None] / n))
    return np.vstack(pts)


def grade(blk, wways, cfg):
    """Return (graded terrain, road mask, splat channel map). Road cross-sections are flat:
    every road pixel takes the smoothed profile height of its nearest centerline pixel."""
    N0, N1 = blk.shape; rc = cfg["roads"]
    center_h = np.full(blk.shape, np.nan, np.float32)
    center_w = np.zeros(blk.shape, np.float32)
    center_c = np.zeros(blk.shape, np.uint8)
    # widest first so narrower roads (links) draw on top at junctions but never shrink a motorway
    for w in sorted(wways, key=lambda w: -rc[w["cls"]][0]):
        width, ch = rc[w["cls"]]
        p = densify(w["xy"], 0.5)
        ix = np.clip(np.round(p[:, 0]).astype(int), 0, N1 - 1); iy = np.clip(np.round(p[:, 1]).astype(int), 0, N0 - 1)
        prof = ndimage.gaussian_filter1d(blk[iy, ix].astype(np.float64), rc["profile_sigma"] * 2, mode="nearest")
        # where a centerline pixel is already set (junction), average -> continuity between roads
        cur = center_h[iy, ix]
        center_h[iy, ix] = np.where(np.isnan(cur), prof, (cur + prof) / 2)
        center_w[iy, ix] = np.maximum(center_w[iy, ix], width)
        center_c[iy, ix] = 1 if ch == "R" else 2
    is_c = ~np.isnan(center_h)
    if not is_c.any():
        return blk, np.zeros(blk.shape, bool), np.zeros(blk.shape, np.uint8)
    dist, (ny, nx) = ndimage.distance_transform_edt(~is_c, return_indices=True)
    half = center_w[ny, nx] / 2
    road = dist <= half
    target = center_h[ny, nx]
    t = np.clip((dist - half) / rc["feather"], 0, 1)                 # 0 on road -> 1 at feather edge
    w = np.where(road, 1.0, 1 - t)
    out = (blk * (1 - w) + np.nan_to_num(target) * w).astype(np.float32)
    ch = np.where(road, center_c[ny, nx], 0).astype(np.uint8)
    return out, road, ch
```

```python
"""Build GeneratedWorlds/<name> from an empty never-loaded RWG shell + our terrain/roads.

python -m tucson.assemble --shell "Tisuviro County" --name Tucson_AZ [--allow-loaded]
"""
import argparse, os, shutil
import numpy as np
from PIL import Image
from . import roads, terrain, zones
from .config import load
from .warp import Warp

GW = os.path.expandvars(r"%APPDATA%\7DaysToDie\GeneratedWorlds")
SHELL_FILES = ["main.ttw", "map_info.xml", "biomes.png", "splat4.png", "radiation.png"]
DERIVED = ("_processed", "_half", "checksums.txt")


def copy_shell(src, dst, allow_loaded=False):
    loaded = [f for f in os.listdir(src) if any(k in f for k in DERIVED)]
    if loaded and not allow_loaded:
        raise RuntimeError(f"shell {src} has been loaded ({loaded[:3]}); pass allow_loaded=True to use it anyway")
    os.makedirs(dst, exist_ok=True)
    for f in SHELL_FILES:
        shutil.copy2(os.path.join(src, f), os.path.join(dst, f))


def write_dtm(path, blk):
    np.round(blk[::-1] * 256).clip(0, 65535).astype("<u2").tofile(path)   # flip: dtm row 0 = south


def write_splat3(path, ch):
    a = np.zeros(ch.shape + (4,), np.uint8)
    a[ch == 1] = [255, 0, 0, 255]; a[ch == 2] = [0, 255, 0, 255]
    Image.fromarray(a, "RGBA").save(path)


def write_spawnpoints(path, pts):
    """pts: list of (x, y, z, yaw_deg) in world coords."""
    body = "".join(f'    <spawnpoint position="{x},{y:.2f},{z}" rotation="0,{yaw},0"/>\n' for x, y, z, yaw in pts)
    open(path, "w", encoding="utf-8").write(f"<spawnpoints>\n{body}</spawnpoints>\n")


def spawn_along_motorway(wways, blk, n=12):
    """n spawn points on motorway centerlines, evenly spread; yaw along the road."""
    N = blk.shape[0]; cands = []
    for w in wways:
        if w["cls"] != "motorway":
            continue
        p = roads.densify(w["xy"], 50)
        for a, b in zip(p[:-1], p[1:]):
            yaw = int(np.degrees(np.arctan2(b[0] - a[0], -(b[1] - a[1]))) % 360)   # 0 = north
            cands.append((int(a[0]), int(a[1]), yaw))
    pick = [cands[i] for i in np.linspace(0, len(cands) - 1, n).astype(int)]
    return [(x - N // 2, float(blk[r, x]) + 1, (N - 1 - r) - N // 2, yaw) for x, r, yaw in pick]


def build(shell, name, allow_loaded=False):
    cfg = load(); warp = Warp(cfg); N = cfg["size"]
    dst = os.path.join(GW, name)
    copy_shell(os.path.join(GW, shell), dst, allow_loaded)
    ele = terrain.elevation_grid(warp, N)
    for zname, rings in zones.polygons(cfg).items():
        ele = terrain.flatten(ele, zones.mask(rings, warp, N))
    blk = terrain.to_blocks(ele, cfg).astype(np.float32)
    wways = roads.to_world(roads.fetch_ways(cfg), warp)
    blk, road, ch = roads.grade(blk, wways, cfg)
    write_dtm(os.path.join(dst, "dtm.raw"), blk)
    write_splat3(os.path.join(dst, "splat3.png"), ch)
    open(os.path.join(dst, "prefabs.xml"), "w", encoding="utf-8").write('<?xml version="1.0" encoding="UTF-8"?>\n<prefabs>\n</prefabs>\n')
    write_spawnpoints(os.path.join(dst, "spawnpoints.xml"), spawn_along_motorway(wways, blk))
    print(f"wrote {dst}: {len(wways)} road pieces, {road.sum()} road px")


if __name__ == "__main__":
    ap = argparse.ArgumentParser(); ap.add_argument("--shell", required=True); ap.add_argument("--name", default="Tucson_AZ")
    ap.add_argument("--allow-loaded", action="store_true"); a = ap.parse_args()
    build(a.shell, a.name, a.allow_loaded)
```
